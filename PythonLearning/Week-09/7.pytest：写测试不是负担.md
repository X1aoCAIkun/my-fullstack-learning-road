import pytest       
from mytodo.core import TaskList            # 导入要测试的部分

def test_add_task():
    t1 = TaskList()
    t = t1.add("写代码")
    assert t.id == 1
    assert t.title == "写代码”

def test_done_invalid_id():
    t1 = TaskList()
    # 这里期望输入错误数据，抛出异常以及报错文字，注意 match 是正则匹配，只有当异常的错误信息里包含match的内容，测试才会通过
    with pytest.raises(ValueError, match="任务不存在"):
        t1.done(999)

# 一组测试用例自动循环跑，"title, ok"就是下面函数的参数名字，后面的用例按位置传入
@pytest.mark.parametrize("title,ok", [
    ("hello", True),
    ("", False),
    ("x" * 200, False)
])
def test_title_validation(title, ok):
    t1 = TaskList()
    if ok:
        t1.add(title)
    else:
        # 添加错误标题时，期望抛出异常
        with pytest.raises(Exception):
            t1.add(title)

# 跑测试文件
uv run pytest -v        # 在虚拟环境里跑， -v是详细输出，会打印每个用例成功/失败
# 覆盖率测试，--cov=mytodo：统计 mytodo 包的代码覆盖率；--cov-report=term-missing：在终端打印报告，标出哪些代码没有测试覆盖到
uv run -pytest --cov=mttodo --cov-report=term-missing   # 覆盖率

# 练习：给 TaskList.done 写 3 个测试：成功、id 不存在、id 是字符串。
# 就是对 TaskList 里的 done 功能的各部分做单元测试
def test_done_ok():
    t1 = TaskList()
    t1.add("x")
    t1.done(1)
    assert t1.tasks[0].status == "done"

def test_done_not_found()
    t1 = TaskList()
    with pytest.raises(ValueError, match="不存在"):
        t1.done(999)

def test_done_str_id():
    t1 = TaskList()
    t1.add("x")
    with pytest.raises(ValueError, match="不存在"):
        t1.done("1")
