# pre-commit 是一个Git钩子工具，在 git commit 自动 ruff check ，不符合规范就阻止提交
uv add --dev pre-commit     # 在本地安装开发依赖

# 配置文件（.yaml文件放在项目根目录，告诉pre-commit要跑的检查工具）
.pre-commit-config.yaml:
repos:
# 第一组：ruff检查代码 + 格式化
    - repo: https://github.com/astral-sh/ruff-pre-commit
      rev: v0.5.0   # 指定ruff这个hook的版本号
      hooks:
        - id: ruff
          args: [--fix]         # 检查代码并修复能自动改的问题
        - id: ruff-format       # 格式化
# 第二组：pre-commit官方自带的钩子
    - repo: https://github.com/pre-commit/pre-commit-hooks
      rev: v4.6.0
      hooks:
        - id: trailing-whitespace       # 删除行末多余空客
        - id: end-of-file-fixer     # 保证文件末尾一个换行符

uv run pre-commit install               # 装 git hook
uv run pre-commit run --all-files       # 全量跑一遍

# 之后每次 git commit, 自动跑 ruff、自动修空白
# repo就是钩子脚本的远程仓库地址；rev固定版本；hooks.id指定要启用哪一个检查脚本
