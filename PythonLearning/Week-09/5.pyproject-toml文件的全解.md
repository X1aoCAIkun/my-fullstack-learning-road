在之前学习了正确的工程化项目目录，本节将对 pyproject.toml 配置文件做全解
它是现代Python项目统一配置文件：打包信息、依赖、工具（ruff/pytest）都写在这


# pyproject
[project]
# PyPI 上的包名（必须唯一，不能重复），这样就可以 pip install mytodo
name = "mytodo"
# 版本号，遵循SemVer（语义化版本）：主.次.补
version = "0.1.0"
# 项目的简短描述           
description = "A tiny CLI todo built with Pydantic " 
# 指定的项目说明文件，上传 PyPI 会展示这个文件内容
readme = "README.md"
# 要求用户本地Python版本必须是3.10+，低于这个版本不能安装
requires-python = ">=3.10"
# 开源协议：MIT协议
license = {text = "MIT"}
# 作者信息
authors = [{name = "Your Name", email = "you@example.com"}]
# 关键词，PyPI 检索用
keywords = ["todo", "cli", "pydantic"]
# 分类标签， 给 PyPI 标记项目环境、许可证
classifiers = [
    "Programming Language :: Python :: 3",
    "License :: OSI Approved :: MIT License",
]
# 运行时必须依赖。别人在 pip install 时和包一起自动安装
dependencies = [
    "pydantic>=2.0",
    "click>=8.0",
]

# 可选依赖分组，这里是开发依赖，可以 pip install ".[dev]"，这是开发者做代码检查、跑测试才需要
[project.optional-dependencies]
dev = ["pytest", "ruff", "mypy"]

# 命令行入口，包名.模块名:函数 -> 导入~包的~模块，执行~函数
[project.scripts]
mytodo = "mytodo.cli:main"          # 命令行入口

# 打包构建系统配置，构建这个包时需要 hatchling工具，使用它的构建后端
[build-system]
# hatchling 现在推荐的打包后端，用来把代码打包成wheel/sdist分发包
requires = ["hatchling"]
build-backend = "hatchling.build"

# [tool.xxx] 不属于打包，只是对应工具自己的配置
[tool.ruff]
# ruff 格式下每行代码的最长字符数
line-length = 100

# pytest 的配置，告诉pytest：测试代码在 tests/文件夹，执行pytest自动去这个目录找测试用例
[tool.pytest.ini_options]
testpaths = ["tests"]

# 注意：project.scripts 是发布后用户能在终端敲 mytodo 而不是 python -m mytodo 的关键
